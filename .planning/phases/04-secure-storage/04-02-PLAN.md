    ```typescript
    /**
     * Session Token Storage
     *
     * Stores a hash of the session token in localStorage for session management.
     * All encryption/decryption is handled server-side by the Edge Function.
     */

    /**
     * Store the session token (hashed) for session management
     */
    export async function storeSessionToken(token: string): Promise<void> {
      try {
        const hash = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(token))
        const hashArray = Array.from(new Uint8Array(hash))
        const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('')
        localStorage.setItem('sb-session-token', hashHex)
      } catch (e) {
        console.warn('Failed to store session token:', e)
      }
    }

    /**
     * Clear the session token on logout
     */
    export function clearSessionToken(): void {
      localStorage.removeItem('sb-session-token')
      sessionStorage.removeItem('sb-session-token')
    }
    ```

    function detectProvider(modelId: string): Provider { /* keep existing logic unchanged */ }

    export async function chatCompletion(systemPrompt: string, userMessage: string, modelOverride?: string): Promise<string> {
      const config = getRuntimeModelConfig()
      const provider = detectProvider(modelOverride || config.chatModel)
      const model = modelOverride || config.chatModel

      try {
        return await proxyChat({ provider, model, systemPrompt, userMessage })
      } catch (primaryError) {
        // Try fallback model if configured
        if (!modelOverride && config.chatFallbackModel && config.chatFallbackProvider) {
          console.warn('Primary chat failed, trying fallback:', primaryError)
          try {
            return await proxyChat({
              provider: config.chatFallbackProvider,
              model: config.chatFallbackModel,
              systemPrompt,
              userMessage,
            })
          } catch {
            throw primaryError  // Throw original error
          }
        }
        throw primaryError
      }
    }

      const config = getRuntimeModelConfig()
      const provider = detectProvider(modelOverride || config.chatModel)
      const model = modelOverride || config.chatModel
      return proxyChatWithImage({ provider, model, systemPrompt, imageUrl })
    }

    export async function textToSpeech(text: string, voice?: string, modelOverride?: string): Promise<string> {
      const config = getRuntimeModelConfig()
      const provider = detectProvider(modelOverride || config.ttsModel)
      const model = modelOverride || config.ttsModel
      const resolvedVoice = voice || config.ttsVoice || 'alloy'

      try {
        return await proxyTTS({ provider, model, voice: resolvedVoice, text })
      } catch (primaryError) {
        if (!modelOverride && config.ttsFallbackModel && config.ttsFallbackProvider) {
          console.warn('Primary TTS failed, trying fallback:', primaryError)
          try {
            return await proxyTTS({
              provider: config.ttsFallbackProvider,
              model: config.ttsFallbackModel,
              voice: resolvedVoice,
              text,
            })
          } catch {
            throw primaryError
          }
        }
        throw primaryError
      }
    }

    export async function speechToText(audioBlob: Blob): Promise<string> {
      const config = getRuntimeModelConfig()
      const provider = detectProvider(config.sttModel)
      const model = config.sttModel

      try {
        return await proxySTT({ provider, model, audioBlob })
      } catch (primaryError) {
        if (config.sttFallbackModel && config.sttFallbackProvider) {
          console.warn('Primary STT failed, trying fallback:', primaryError)
          try {
            return await proxySTT({
              provider: config.sttFallbackProvider,
              model: config.sttFallbackModel,
              audioBlob,
            })
          } catch {
            throw primaryError
          }
        }
        throw primaryError
      }
    }

      size?: string
      quality?: string
      style?: string
      n?: number
    }

      const config = getRuntimeModelConfig()
