function detectSource(modelId: string): Source {
  if (modelId.startsWith('gemini')) return 'genai';
  if (
    modelId.startsWith('llama-') ||
    modelId.startsWith('meta-llama/') ||
    modelId.startsWith('qwen/') ||
    modelId.startsWith('canopylabs/') ||
    modelId.startsWith('whisper-large-v3')
  ) return 'groq';
  if (modelId.includes('/')) return 'openrouter';
  return 'openai';
}

| Client -> model catalog | Model IDs from runtime config or overrides cross into catalog lookup |
| Catalog -> proxy call | Resolved source is passed to aiProxy which routes to external AI providers |

| Threat ID | Category | Component | Disposition | Mitigation Plan |
